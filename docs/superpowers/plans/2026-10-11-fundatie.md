# Fundație (etapele 0–1) — Plan de implementare

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Un proiect Astro care se construiește, se testează și se verifică în CI, cu tokens de brand, layout de bază, header, footer, primul Web Component (FAQ) și un schelet de server Node care rulează din bundle.

**Architecture:** Astro generează HTML static. Interactivitatea vine din Web Components în light DOM, care îmbunătățesc HTML-ul deja randat. Serverul e un handler `Request → Response`, împachetat cu esbuild într-un singur fișier pentru Passenger. Granițele dintre straturi sunt verificate automat: dependency-cruiser pentru `.ts`, ESLint pentru `.astro` (dependency-cruiser nu citește `.astro`).

**Tech Stack:** Astro 7.3.8, TypeScript ~6.0, Vitest 5, happy-dom, esbuild, ESLint 10 + typescript-eslint + eslint-plugin-astro, dependency-cruiser 18, @fontsource (Nunito, Barlow Condensed), Node ^22.22.3 || ≥ 24.16.0.

**Spec:** `docs/superpowers/specs/2026-10-09-site-englishminds-design.md` (citește și `.claude/skills/project-conventions/SKILL.md`)

`src/shared/content/find-todo-markers.ts`:

```ts
export function findTodoMarkers(value: unknown, path = ''): string[] {
  if (typeof value === 'string') return value.includes('TODO') ? [path] : [];
  if (Array.isArray(value)) {
    return value.flatMap((item, index) => findTodoMarkers(item, `${path}[${index}]`));
  }
  if (value !== null && typeof value === 'object') {
    return Object.entries(value).flatMap(([key, item]) =>
      findTodoMarkers(item, path ? `${path}.${key}` : key),
    );
  }
  return [];
}
```

- [ ] **Step 2: Implementează `environment.ts`**

`server/config/environment.ts`:

```ts
export type Environment = Readonly<{ port: number }>;

export function readEnvironment(env: NodeJS.ProcessEnv = process.env): Environment {
  const raw = env.PORT ?? '3000';
  const port = Number(raw);
  if (raw.trim() === '' || !Number.isInteger(port) || port < 1 || port > 65535) {
    throw new Error(`PORT invalid: "${raw}"`);
  }
  return Object.freeze({ port });
}
```

```ts
import { createServer, type IncomingMessage, type ServerResponse } from 'node:http';
import { Readable } from 'node:stream';
import { readEnvironment } from './config/environment';
import { handleRequest } from './handle-request';

function toWebRequest(req: IncomingMessage): Request {
  const headers = new Headers();
  for (const [name, value] of Object.entries(req.headers)) {
    if (Array.isArray(value)) value.forEach((item) => headers.append(name, item));
    else if (value !== undefined) headers.set(name, value);
  }
  const hasBody = req.method !== 'GET' && req.method !== 'HEAD';
  const init: RequestInit & { duplex: 'half' } = {
    method: req.method,
    headers,
    body: hasBody ? (Readable.toWeb(req) as ReadableStream<Uint8Array>) : undefined,
    duplex: 'half',
  };
  return new Request(new URL(req.url ?? '/', 'http://localhost'), init);
}

async function sendWebResponse(response: Response, res: ServerResponse): Promise<void> {
  res.statusCode = response.status;
  response.headers.forEach((value, name) => res.setHeader(name, value));
  res.end(Buffer.from(await response.arrayBuffer()));
}

const { port } = readEnvironment();

createServer((req, res) => {
  handleRequest(toWebRequest(req))
    .then((response) => sendWebResponse(response, res))
    .catch((error: unknown) => {
      console.error('[server] unhandled request error', { method: req.method, url: req.url, error });
      res.statusCode = 500;
      res.setHeader('content-type', 'application/json; charset=utf-8');
      res.end(JSON.stringify({ ok: false, error: 'internal' }));
    });
}).listen(port, () => {
  console.log(`[server] listening on ${port}`);
});
```

`src/features/faq/ui/faq.element.ts`:

```ts
export class FaqList extends HTMLElement {
  connectedCallback(): void {
    if (this.dataset.enhanced === 'true') return;
    this.dataset.enhanced = 'true';
    for (const item of this.items()) {
      const question = item.querySelector<HTMLElement>('.faq__question');
      const answer = item.querySelector<HTMLElement>('.faq__answer');
      if (!question || !answer) continue;
      const button = document.createElement('button');
      button.type = 'button';
      button.className = 'faq__toggle';
      button.setAttribute('aria-expanded', 'false');
      button.setAttribute('aria-controls', answer.id);
      button.append(...question.childNodes);
      question.append(button);
      answer.hidden = true;
      button.addEventListener('click', () => this.toggle(item));
    }
  }

  private items(): HTMLElement[] {
    return [...this.querySelectorAll<HTMLElement>(':scope > .faq__item')];
  }

  private toggle(target: HTMLElement): void {
    const wasOpen = target.querySelector('.faq__toggle')?.getAttribute('aria-expanded') === 'true';
    for (const item of this.items()) {
      const open = item === target && !wasOpen;
      item.querySelector('.faq__toggle')?.setAttribute('aria-expanded', String(open));
      const answer = item.querySelector<HTMLElement>('.faq__answer');
      if (answer) answer.hidden = !open;
    }
  }
}

export function defineFaqList(): void {
  if (!customElements.get('faq-list')) customElements.define('faq-list', FaqList);
}
```
